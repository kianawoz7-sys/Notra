cmake_minimum_required(VERSION 3.22.1)
project(catatvoice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_FLAGS_DEBUG "-O3 -g0" CACHE STRING "Optimized offline inference" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g0" CACHE STRING "Optimized offline inference" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" "whisper-build")
add_library(catatvoice SHARED voice.cpp)
target_link_libraries(catatvoice PRIVATE whisper android log)

# Android ARM baseline remains available. Use a separately linked CPU backend
# with FP16 and dot-product kernels only when /proc/cpuinfo advertises both.
if(ANDROID_ABI STREQUAL "arm64-v8a")
    get_target_property(cpu_sources ggml-cpu SOURCES)
    get_target_property(cpu_source_dir ggml-cpu SOURCE_DIR)
    get_target_property(cpu_includes ggml-cpu INCLUDE_DIRECTORIES)
    get_target_property(cpu_definitions ggml-cpu COMPILE_DEFINITIONS)
    set(fast_sources "")
    foreach(cpu_source IN LISTS cpu_sources)
        if(IS_ABSOLUTE "${cpu_source}")
            list(APPEND fast_sources "${cpu_source}")
        else()
            list(APPEND fast_sources "${cpu_source_dir}/${cpu_source}")
        endif()
    endforeach()
    add_library(catat-cpu-fast STATIC ${fast_sources})
    target_include_directories(catat-cpu-fast PRIVATE ${cpu_includes})
    if(cpu_definitions)
        target_compile_definitions(catat-cpu-fast PRIVATE ${cpu_definitions})
    endif()
    target_compile_options(catat-cpu-fast PRIVATE -O3 -march=armv8.2-a+fp16+dotprod)
    target_compile_features(catat-cpu-fast PRIVATE c_std_11 cxx_std_17)
    target_link_libraries(catat-cpu-fast PRIVATE ggml-base)
    add_library(catatvoice_fast SHARED voice.cpp)
    # Resolve whisper's CPU symbols from the fast archive before the generic
    # transitive backend. Each JNI library contains its own complete context.
    target_link_libraries(catatvoice_fast PRIVATE whisper catat-cpu-fast android log)
endif()
